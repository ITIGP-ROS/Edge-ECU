/* ============================================================
 *  CRC8 — poly 0x07, init 0x00 (SMBUS)
 * ============================================================ */
static uint8_t crc8(const uint8_t *data, uint16_t len)
{
    uint8_t crc = 0x00U;
    for (uint16_t i = 0U; i < len; i++)
    {
        crc ^= data[i];
        for (uint8_t b = 0U; b < 8U; b++)
            crc = (crc & 0x80U) ? (uint8_t)((crc << 1U) ^ 0x07U)
                                 : (uint8_t)(crc << 1U);
    }
    return crc;
}

static void delay(volatile uint32_t n) { while (n-- > 0U); }

/* ============================================================
 *  Test result storage — read via debugger Watch panel
 * ============================================================ */
#define LOG_SIZE 1024U
static char     test_log[LOG_SIZE];
static uint16_t log_pos   = 0U;
static uint8_t  pass_count = 0U;
static uint8_t  fail_count = 0U;

static void log_str(const char *s)
{
    while (*s && log_pos < (LOG_SIZE - 1U))
        test_log[log_pos++] = *s++;
    test_log[log_pos] = '\0';
}

static void log_uint(uint32_t v)
{
    if (v == 0U) { log_str("0"); return; }
    char buf[10]; uint8_t i = 0U;
    while (v > 0U) { buf[i++] = (char)('0' + (v % 10U)); v /= 10U; }
    while (i > 0U) { char c[2] = {buf[--i], '\0'}; log_str(c); }
}

static void report(const char *name, uint8_t pass)
{
    log_str(pass ? "PASS: " : "FAIL: ");
    log_str(name);
    log_str("\n");
    if (pass) pass_count++; else fail_count++;
}

/* ============================================================
 *  GPIO setup
 * ============================================================ */
static void config_uart1_gpio(void)
{
    RCC_EN_CLK_PERIPHERAL(PERIPH_GPIOA);

    GPIO_CONFIG_t tx = {
        .Pin = GPIO_PIN9, .Port = GPIO_PORTA,
        .Mode = GPIO_MODE_ALTERNATE, .Type = GPIO_OUTPUT_PUSH_PULL,
        .Speed = GPIO_SPEED_VERY_HIGH, .Pull = GPIO_NO_PULL,
        .Alternate = AF_USART_1_2
    };
    GPIO_INIT(&tx);

    GPIO_CONFIG_t rx = {
        .Pin = GPIO_PIN10, .Port = GPIO_PORTA,
        .Mode = GPIO_MODE_ALTERNATE, .Type = GPIO_OUTPUT_PUSH_PULL,
        .Speed = GPIO_SPEED_VERY_HIGH, .Pull = GPIO_PULL_UP,
        .Alternate = AF_USART_1_2
    };
    GPIO_INIT(&rx);
}

/* ============================================================
 *  MAIN
 * ============================================================ */
int main(void)
{
    /* ---- Boot sequence ---- */
    RCC_INIT_84MHz_HSI();
    RCC_EN_CLK_PERIPHERAL(PERIPH_USART1);
    RCC_EN_CLK_PERIPHERAL(PERIPH_DMA2);   /* DMA2 clock must be enabled */
    config_uart1_gpio();

    /* ---- UART1 init — DMA TX mode ---- */
    UART_Config_t uart_cfg = {
        .uart_id      = UART1_ID,
        .baudrate     = 921600U,
        .parity       = UART_PARITY_NONE,
        .stop_bits    = UART_STOP_1,
        .data_bits    = UART_DATA_8BIT,
        .oversampling = UART_OVER8_DISABLE,
        .tx_mode      = UART_MODE_DMA,     /* DMA TX */
        .rx_mode      = UART_MODE_IRQ      /* IRQ RX ring buffer */
    };
    UART_Error_t uart_ret = UART_Init(&uart_cfg);
    report("UART_Init DMA mode", uart_ret == UART_OK);

    /* ---- Service layer init — DMA2 Stream7 Channel4 for UART1 TX ---- */
    UART_SVC_Error_t svc_ret = UART_SVC_Init(
        UART1_ID,
        5U,                        /* UART NVIC priority */
        UART_SVC_TX_MODE_DMA,
        DMA_2,                     /* DMA2 */
        DMA_STREAM_7,              /* Stream 7 */
        DMA_CHANNEL_4              /* Channel 4 — USART1_TX per RM0368 Table 27 */
    );
    report("UART_SVC_Init DMA2 S7 Ch4", svc_ret == UART_SVC_OK);

    log_str("========================================\n");
    log_str("DMA TX bench test — 921600 8N1\n");
    log_str("========================================\n");

    /* ============================================================
     * D-T01: Single frame DMA TX — known 11-byte frame
     * Format: [0xAA][0x55][TYPE][LEN][P0..P5][CRC8]
     * Visible on terminal as binary — use hexlify filter:
     *   pio device monitor --baud 921600 --filter hexlify
     * Expected: DMA_SVC_TransmitDMA returns UART_SVC_OK
     *           tx_active goes 1 then back to 0 after TC
     * ============================================================ */
    static uint8_t  frame_raw[11];
    static Buffer_t frame = BUFFER_INIT(frame_raw, 11);

    frame.data[0]  = 0xAAU;          /* sync byte 1 */
    frame.data[1]  = 0x55U;          /* sync byte 2 */
    frame.data[2]  = 0x01U;          /* TYPE = sensor data */
    frame.data[3]  = 0x06U;          /* LEN  = 6 payload bytes */
    frame.data[4]  = 0x10U;          /* payload[0] */
    frame.data[5]  = 0x20U;          /* payload[1] */
    frame.data[6]  = 0x30U;          /* payload[2] */
    frame.data[7]  = 0x40U;          /* payload[3] */
    frame.data[8]  = 0x50U;          /* payload[4] */
    frame.data[9]  = 0x60U;          /* payload[5] */
    frame.data[10] = crc8(frame.data, 10U);  /* CRC over bytes 0–9 */
    frame.length   = 11U;

    svc_ret = UART_SVC_TransmitDMA(UART1_ID, &frame);
    report("D-T01 TransmitDMA single frame returns OK",
           svc_ret == UART_SVC_OK);

    /* Wait for DMA to complete — poll IsTxIdle with timeout */
    uint32_t wait = 420000U;  /* ~5ms at 84MHz — 11 bytes at 921600 takes ~120µs */
    uint8_t  idle = 0U;
    while (wait-- > 0U)
    {
        UART_SVC_IsTxIdle(UART1_ID, &idle);
        if (idle == 1U) break;
    }
    report("D-T01 DMA completes — tx_active back to 0", idle == 1U);

    /* ============================================================
     * D-T02: Busy rejection — call TransmitDMA twice without waiting
     * First call must succeed, second must return UART_SVC_ERROR_BUSY
     * ============================================================ */
    static uint8_t  frame2_raw[4];
    static Buffer_t frame2 = BUFFER_INIT(frame2_raw, 4);
    frame2.data[0] = 0xAAU;
    frame2.data[1] = 0x55U;
    frame2.data[2] = 0x02U;
    frame2.data[3] = crc8(frame2.data, 3U);
    frame2.length  = 4U;

    /* Start first transfer */
    svc_ret = UART_SVC_TransmitDMA(UART1_ID, &frame2);
    report("D-T02 first call OK", svc_ret == UART_SVC_OK);

    /* Immediately try second — should be rejected */
    UART_SVC_Error_t busy_ret = UART_SVC_TransmitDMA(UART1_ID, &frame2);
    report("D-T02 second call rejected BUSY",
           busy_ret == UART_SVC_ERROR_BUSY);

    /* Wait for first to finish */
    wait = 420000U; idle = 0U;
    while (wait-- > 0U)
    {
        UART_SVC_IsTxIdle(UART1_ID, &idle);
        if (idle == 1U) break;
    }
    report("D-T02 first transfer completes after busy reject", idle == 1U);

    /* ============================================================
     * D-T03: Invalid parameter rejection
     * NULL buf, zero length, bad ID
     * ============================================================ */
    svc_ret = UART_SVC_TransmitDMA(UART1_ID, NULL);
    report("D-T03 NULL buf rejected", svc_ret == UART_SVC_ERROR_PARAM);

    static uint8_t  empty_raw[4];
    static Buffer_t empty_buf = BUFFER_INIT(empty_raw, 4);
    empty_buf.length = 0U;
    svc_ret = UART_SVC_TransmitDMA(UART1_ID, &empty_buf);
    report("D-T03 zero length rejected", svc_ret == UART_SVC_ERROR_PARAM);

    svc_ret = UART_SVC_TransmitDMA((UART_Id_t)99U, &frame);
    report("D-T03 bad ID rejected", svc_ret == UART_SVC_ERROR_PARAM);

    /* ============================================================
     * D-T04: 1000-frame DMA stress test
     * Send 1000 CRC8-framed packets back to back
     * Each frame: [0xAA][0x55][SEQ_LO][SEQ_HI][CRC]
     * Wait for idle between each frame
     * Expected: all 1000 complete, zero failures, zero timeouts
     * ============================================================ */
    log_str("D-T04: 1000-frame DMA stress...\n");

    static uint8_t  stress_raw[5];
    static Buffer_t stress = BUFFER_INIT(stress_raw, 5);

    uint16_t stress_fail    = 0U;
    uint16_t stress_timeout = 0U;

    for (uint16_t seq = 0U; seq < 1000U; seq++)
    {
        stress.data[0] = 0xAAU;
        stress.data[1] = 0x55U;
        stress.data[2] = (uint8_t)(seq & 0xFFU);
        stress.data[3] = (uint8_t)((seq >> 8U) & 0xFFU);
        stress.data[4] = crc8(stress.data, 4U);
        stress.length  = 5U;

        svc_ret = UART_SVC_TransmitDMA(UART1_ID, &stress);
        if (svc_ret != UART_SVC_OK)
        {
            stress_fail++;
            continue;
        }

        /* Wait for completion before sending next frame */
        wait = 840000U;  /* ~10ms timeout per frame */
        idle = 0U;
        while (wait-- > 0U)
        {
            UART_SVC_IsTxIdle(UART1_ID, &idle);
            if (idle == 1U) break;
        }
        if (idle == 0U) stress_timeout++;
    }

    report("D-T04 1000 frames no TX error",    stress_fail    == 0U);
    report("D-T04 1000 frames no timeout",      stress_timeout == 0U);
    if (stress_fail > 0U)
    { log_str("  TX errors: "); log_uint(stress_fail); log_str("\n"); }
    if (stress_timeout > 0U)
    { log_str("  Timeouts:  "); log_uint(stress_timeout); log_str("\n"); }

    /* ============================================================
     * D-T05: DMA transfer status after completion
     * After a completed transfer DMA_GetTransferStatus should
     * return DMA_TRANSFER_COMPLETE (flags not cleared until next start)
     * ============================================================ */
    stress.data[0] = 0xAAU;
    stress.data[1] = 0x55U;
    stress.data[2] = 0xFFU;
    stress.data[3] = 0xFFU;
    stress.data[4] = crc8(stress.data, 4U);
    stress.length  = 5U;

    UART_SVC_TransmitDMA(UART1_ID, &stress);
    wait = 420000U; idle = 0U;
    while (wait-- > 0U)
    {
        UART_SVC_IsTxIdle(UART1_ID, &idle);
        if (idle == 1U) break;
    }

    DMA_TransferStatus_t dma_status;
    DMA_Error_t dma_ret = DMA_GetTransferStatus(DMA_2, DMA_STREAM_7, &dma_status);
    report("D-T05 DMA_GetTransferStatus returns OK", dma_ret == DMA_OK);
    report("D-T05 Status is COMPLETE after transfer",
           dma_status == DMA_TRANSFER_COMPLETE);

    /* ============================================================
     * Summary
     * ============================================================ */
    log_str("========================================\n");
    log_str("PASS: "); log_uint(pass_count); log_str("\n");
    log_str("FAIL: "); log_uint(fail_count); log_str("\n");
    log_str("========================================\n");

    /* Heartbeat */
    RCC_EN_CLK_PERIPHERAL(PERIPH_GPIOC);
    GPIO_CONFIG_t led = {
        .Pin = GPIO_PIN13, .Port = GPIO_PORTC,
        .Mode = GPIO_MODE_OUTPUT, .Type = GPIO_OUTPUT_PUSH_PULL,
        .Speed = GPIO_SPEED_LOW, .Pull = GPIO_NO_PULL,
        .Alternate = AF_SYSTEM
    };
    GPIO_INIT(&led);

    while (1)
    {
        GPIO_TogglePin(GPIO_PORTC, GPIO_PIN13);
        delay(840000U);
    }

    return 0U;
}
