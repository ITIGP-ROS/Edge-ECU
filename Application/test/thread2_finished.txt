/* Latest sensor sample written by DMA-done callback, read by task */
static volatile MPU6050_RawData_t s_latest_sample;


/* ============== ISR-context callback ============== */
static void on_tim2_update(void *ctx)
{
    (void)ctx;
    BaseType_t woken = pdFALSE;
    xSemaphoreGiveFromISR(s_tim_sem, &woken);
    portYIELD_FROM_ISR(woken);
}
/* ============== MPU6050 DMA-done callback (ISR context) ============== */
static void on_mpu_read_done(const MPU6050_RawData_t *data, void *ctx)
{
    (void)ctx;
    s_latest_sample = *data;        /* copy into module state */
    BaseType_t woken = pdFALSE;
    xSemaphoreGiveFromISR(s_dma_sem, &woken);
    portYIELD_FROM_ISR(woken);
}

/* ============== Task body ============== */
static void Thread1_Sensor(void *arg)
{
    (void)arg;
    MPU6050_RawData_t snapshot;

    for (;;)
    {
        /* 1. Wait 10ms tick from TIM2 */
        xSemaphoreTake(s_tim_sem, portMAX_DELAY);

        /* 2. Kick off DMA read */
        if (MPU6050_TriggerRead(on_mpu_read_done, NULL) == MPU6050_OK)
        {
            /* 3. Wait for DMA done */
            if (xSemaphoreTake(s_dma_sem, pdMS_TO_TICKS(5)) == pdTRUE)
            {
                /* 4. Snapshot and push */
                taskENTER_CRITICAL();
                snapshot = (MPU6050_RawData_t)s_latest_sample;
                taskEXIT_CRITICAL();

                if (RingBuffer_Push(&snapshot) == RING_BUFFER_OK)
                {
                    /* 5. NEW: Notify Thread 2 that data is available */
                    xTaskNotifyGive(s_thread2_handle);
                }
                
                GPIO_TogglePin(GPIO_PORTC, GPIO_PIN13);
            }
        }
    }
}


static void Thread2_TinyML(void *arg)
{
    (void)arg;

    /* Window buffer — lives on stack, only used inside the inner loop */
    MPU6050_RawData_t window[WINDOW_SIZE];      /* 50 × 14B = 700B on stack */

    /* Working buffers — static so they don't bloat the stack */
    static float32_t  scaled[WINDOW_SIZE][N_FEATURES];
    static float32_t  features[N_STAT_FEATURES];
    static int8_t     ts_q[WINDOW_SIZE * N_FEATURES];
    static int8_t     stat_q[N_STAT_FEATURES];

    /* Initialize ML modules once */
    Vote_t vote;
    Vote_Init(&vote);
    Features_Init();
    Inference_Init();

    for (;;)
    {
        /* 1. Sleep until Thread 1 notifies us */
        ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
        g_t2_wake_count++;

        /* 2. Drain ALL available windows in one wake-up
         *    (might be 0 windows if we woke too early — that's fine) */
        while (RingBuffer_Count() >= WINDOW_SIZE)
        {
            /* 3. Snapshot the window */
            if (RingBuffer_PeekWindow(window, WINDOW_SIZE) != RING_BUFFER_OK)
            {
                break;   /* Should not happen given the Count check, but be safe */
            }

            /* 4. ML pipeline */
            Scale_RawWindow((const int16_t (*)[N_FEATURES])window, scaled);

            /* After Scale_RawWindow */

            Features_Extract(scaled, features);
            Quantize_TS(scaled, ts_q);
            Quantize_NormalizeStat(features);
            Quantize_Stat(features, stat_q);





            Inference_Result_t result;
            Inference_Run(ts_q, stat_q, &result);
            g_t2_inference_count++;

            /* 5. Vote */
            Vote_Push(&vote, result.label);
            if (Vote_Ready(&vote))
            {
                result.label = Vote_Decide(&vote);
                
                /* 6. Send to Thread 3 — non-blocking, drop if full */
                if (xQueueSend(g_result_queue, &result, 0) != pdTRUE)
                {
                    g_t2_queue_drops++;
                }
            }

            /* 7. Slide window forward by STRIDE samples */
            RingBuffer_Advance(WINDOW_STRIDE);
        }
    }
}



/* ============== FreeRTOS static hooks (mandatory) ============== */
void vApplicationGetIdleTaskMemory(StaticTask_t **tcb, StackType_t **stack, uint32_t *size)
{
    static StaticTask_t t;
    static StackType_t  s[configMINIMAL_STACK_SIZE];
    *tcb = &t; *stack = s; *size = configMINIMAL_STACK_SIZE;
}


void vApplicationStackOverflowHook(TaskHandle_t x, char *n) { (void)x; (void)n; for(;;); }
void vApplicationIdleHook(void)
{
    static uint32_t count = 0;
    count++;

    if (count % 500000 == 0)
    {
        TaskStatus_t xStatus;

        vTaskGetInfo(s_thread1_handle,
                     &xStatus,
                     pdTRUE,
                     eInvalid);

        /* Raw */
        g_sensor_cpu_cycles      = xStatus.ulRunTimeCounter;
        g_sensor_stack_remaining = xStatus.usStackHighWaterMark;
        g_total_runtime          = ulGetRunTimeCounterValue();

        /* Calculated */                                         /* NEW */
        if (g_total_runtime > 0)                                 /* NEW */
        {                                                        /* NEW */
            g_sensor_cpu_percent_x100 =                          /* NEW */
                (g_sensor_cpu_cycles * 10000U) / g_total_runtime;/* NEW */
            /* x100 format: 350 means 3.50%, 16 means 0.16% */  /* NEW */
        }                                                        /* NEW */
                                                                 /* NEW */
        g_sensor_stack_used    = 256U - g_sensor_stack_remaining;/* NEW */
        g_sensor_stack_percent =                                 /* NEW */
            (g_sensor_stack_remaining * 100U) / 256U;            /* NEW */
        g_uptime_ms = g_total_runtime / (84000000U / 1000U);    /* NEW */
    }
}

static void config_gpio_i2c1(void)
{
    /* PB6 = SCL, PB7 = SDA, AF4, open-drain, pull-up */
    GPIO_CONFIG_t scl = {
        .Pin = GPIO_PIN6, .Port = GPIO_PORTB,
        .Mode = GPIO_MODE_ALTERNATE, .Type = GPIO_OUTPUT_OPEN_DRAIN,
        .Speed = GPIO_SPEED_HIGH, .Pull = GPIO_PULL_UP,
        .Alternate = AF_I2C_1_3
    };
    GPIO_INIT(&scl);

    GPIO_CONFIG_t sda = {
        .Pin = GPIO_PIN7, .Port = GPIO_PORTB,
        .Mode = GPIO_MODE_ALTERNATE, .Type = GPIO_OUTPUT_OPEN_DRAIN,
        .Speed = GPIO_SPEED_HIGH, .Pull = GPIO_PULL_UP,
        .Alternate = AF_I2C_1_3
    };
    GPIO_INIT(&sda);
}

static void config_led_pc13(void)
{
    GPIO_CONFIG_t led = {
        .Pin = GPIO_PIN13, .Port = GPIO_PORTC,
        .Mode = GPIO_MODE_OUTPUT, .Type = GPIO_OUTPUT_PUSH_PULL,
        .Speed = GPIO_SPEED_LOW, .Pull = GPIO_NO_PULL,
        .Alternate = AF_SYSTEM
    };
    GPIO_INIT(&led);
}


static void app_i2c_init(void)
{
    I2C_Config_t    i2c_cfg;
    
    I2C_SVC_Error_t svc_err;

    i2c_cfg.id                = I2C_ID_1;
    i2c_cfg.speed             = I2C_SPEED_FAST;           /* 400 kHz        */
    i2c_cfg.duty_cycle        = I2C_DUTY_2;               /* Tlow/Thigh=2   */
    i2c_cfg.addressing_mode   = I2C_ADDR_7BIT;
    i2c_cfg.own_address1      = 0x00U;
    i2c_cfg.own_address2      = 0x00U;
    i2c_cfg.dual_address_mode = I2C_DUAL_ADDR_DISABLE;
    i2c_cfg.general_call_mode = I2C_GENERAL_CALL_DISABLE;
    i2c_cfg.clock_stretching  = I2C_CLOCK_STRETCH_ENABLE;
    i2c_cfg.analog_filter     = I2C_ANALOG_FILTER_ENABLE;
    i2c_cfg.digital_filter    = I2C_DIGITALFILTER_0;
    i2c_cfg.transfer_mode     = I2C_MODE_DMA;

    
    
     i2c_err = I2C_Init(&i2c_cfg);
    if (i2c_err != I2C_OK) { while (1); }
    

    svc_err = I2C_SVC_Init(
        I2C_ID_1,
        6U,
        6U,
        DMA_1,
        DMA_STREAM_0,
        DMA_CHANNEL_1,
        6U
    );
    




}

/* ============== main ============== */
int main(void)
{
    
   /* 1. Clocks */
    RCC_INIT_84MHz_HSI();

    

    RCC_EN_CLK_PERIPHERAL(PERIPH_GPIOB);
    RCC_EN_CLK_PERIPHERAL(PERIPH_GPIOC);
    RCC_EN_CLK_PERIPHERAL(PERIPH_I2C1);
    RCC_EN_CLK_PERIPHERAL(PERIPH_DMA1);

    /* 2. GPIO */
   // app_i2c_bus_recovery();
    config_gpio_i2c1();
    app_i2c_init();
    config_led_pc13();
    

    

   /* Read BASEPRI */
uint32_t basepri_val;
__asm volatile ("mrs %0, basepri" : "=r" (basepri_val));
volatile uint32_t dbg_basepri = basepri_val;

__asm volatile("nop");  /* breakpoint here */
   
    /* 5. MPU6050 init */
    MPU6050_Init(I2C_ID_1, (I2C_DevAddr7_t)MPU6050_ADDR_LOW, 5000000UL);

    /* 6. Ring buffer */
    RingBuffer_Init();



    /* 3. Init TIM2 with callback */
    TIM_Config_t cfg = {
        .id        = TIM_ID_2,
        .prescaler = TIM_PSC_100HZ,
        .period    = TIM_ARR_100HZ,
        .mode      = TIM_MODE_UP,
        .arpe      = TIM_ARPE_ENABLE,
        .callback  = on_tim2_update,
        .ctx       = NULL
    };
    TIM_Init(&cfg);
    

       /* 2. Create semaphore */
     /* Semaphores for Thread 1 */
    s_tim_sem = xSemaphoreCreateBinaryStatic(&s_tim_sem_storage);
    s_dma_sem = xSemaphoreCreateBinaryStatic(&s_dma_sem_storage);

    /* Queue for Thread 2 → Thread 3 */
    g_result_queue = xQueueCreateStatic(
        4U,                              /* depth */
        sizeof(Inference_Result_t),
        s_result_queue_buffer,
        &s_result_queue_storage
    );
    configASSERT(g_result_queue != NULL);

    /* 4. NVIC: priority 5 (FreeRTOS-safe), then enable */
    NVIC_SetPriority(TIM2, 5U);
    NVIC_EnableIRQ(TIM2);

    /* 5. Start timer */
    TIM_Start(TIM_ID_2);
    
     /* IMPORTANT: Create Thread 2 BEFORE Thread 1
     * because Thread 1 will call xTaskNotifyGive(s_thread2_handle)
     * and we need that handle to be valid before Thread 1 runs. */
    s_thread2_handle = xTaskCreateStatic(
        Thread2_TinyML, "ML", 1024, NULL, 2,
        s_thread2_stack, &s_thread2_tcb
    );
    configASSERT(s_thread2_handle != NULL);

    s_thread1_handle = xTaskCreateStatic(
        Thread1_Sensor, "Sensor", 256, NULL, 3,
        s_thread1_stack, &s_thread1_tcb
    );
    configASSERT(s_thread1_handle != NULL);

    /* PHASE 4: Hand over to scheduler */
    vTaskStartScheduler();
    for (;;) {}
}
